### 4.ST3GAL4阳性与阴性CAF差异表达及GSEA富集分析
## 4.1 划分ST3GAL4高低组并计算差异基因
DefaultAssay(klw_caf) = "RNA"

## 4.2 构建排序向量与多数据库GSEA富集 (C5 / Hallmark / C2)
gene_rank = deg %>% filter(!is.na(avg_log2FC)) %>% distinct(gene, .keep_all = TRUE)
ranks     = sort(setNames(gene_rank$avg_log2FC, gene_rank$gene), decreasing = TRUE)

geneset_c5  = msigdbr(species = "Homo sapiens", category = "C5") %>% dplyr::select(gs_name, gene_symbol)
geneset_h   = msigdbr(species = "Homo sapiens", category = "H")  %>% dplyr::select(gs_name, gene_symbol)
geneset_c2  = msigdbr(species = "Homo sapiens", category = "C2") %>% dplyr::select(gs_name, gene_symbol)
geneset_all = bind_rows(geneset_c5, geneset_h, geneset_c2) %>% distinct()

gsea_res = GSEA(geneList = ranks, TERM2GENE = geneset_all, pvalueCutoff = 1, minGSSize = 10, maxGSSize = 500, verbose = FALSE)
gsea_df  = as.data.frame(gsea_res)
write.csv(gsea_df, "ST3GAL4.high_CAF_C5_GSEA_GSE125449.csv", row.names = FALSE)

## 4.3 批量绘制目标通路GSEA富集曲线
pathways = c(
  "GOBP_EPITHELIAL_TO_MESENCHYMAL_TRANSITION",
  "GOBP_TGF_BETA_SIGNALING",
  "GOBP_INFLAMMATORY_RESPONSE",
  "GOBP_CHEMOKINE_MEDIATED_SIGNALING_PATHWAY",
  "GOBP_RESPONSE_TO_INTERFERON_GAMMA",
  "GOBP_ACTIVATION_OF_IMMUNE_RESPONSE",
  "GOBP_IMMUNE_RESPONSE",
  "GOBP_MYELOID_LEUKOCYTE_MIGRATION",
  "GOBP_MYELOID_LEUKOCYTE_DIFFERENTIATION",
  "GOBP_EXTRACELLULAR_MATRIX_ASSEMBLY",
  "GOCC_COLLAGEN_CONTAINING_EXTRACELLULAR_MATRIX",
  "GOMF_EXTRACELLULAR_MATRIX_STRUCTURAL_CONSTITUENT",
  "HALLMARK_EPITHELIAL_MESENCHYMAL_TRANSITION",
  "HALLMARK_HYPOXIA",
  "HALLMARK_APOPTOSIS",
  "NABA_ECM_GLYCOPROTEINS",
  "HALLMARK_COAGULATION",
  "HALLMARK_COMPLEMENT"
)

p_col = intersect(c("p.adjust", "pvalue", "qvalue"), colnames(gsea_res@result))[1]
if (is.na(p_col)) p_col = "pvalue"

for (pw in pathways) {
  if (pw %in% gsea_res@result$ID) {
    stat  = gsea_res@result[gsea_res@result$ID == pw, ]
    p_val = if (!is.na(stat[[p_col]])) sprintf("%.3e", stat[[p_col]]) else "NA"
    
    p.GSEA = gseaplot2(
      gsea_res,
      geneSetID = pw,
      color     = "#FA5860",
      title     = paste0("CAF GSEA: ", pw, "\nNES = ", round(stat$NES, 2), ", ", p_col, " = ", p_val)
    )
    pdf(paste0("GSEA_", pw, ".pdf"), width = 7, height = 5)
    print(p.GSEA)
    dev.off()
  }
}
