## 1.1 构建实验设计矩阵与对比模型
group  = phenodata[, 1]
design = model.matrix(~ 0 + factor(group))
rownames(design) = colnames(datExpr)
colnames(design) = levels(factor(group))

contrast.matrix = makeContrasts("Ctrl-CE", levels = design)

## 1.2 定义并运行limma差异分析函数
run.limma.deg = function(datExpr, design, contrast.matrix) {
  fit        = lmFit(datExpr, design)
  fit2       = contrasts.fit(fit, contrast.matrix)
  fit2       = eBayes(fit2)
  tempOutput = topTable(fit2, coef = 1, n = Inf)
  nrDEG      = na.omit(tempOutput)
  return(nrDEG)
}

deg = run.limma.deg(datExpr, design, contrast.matrix)

## 1.3 标记上下调显著性并保存结果
deg$sig = "no"
deg$sig[deg$P.Value <= 0.05 & deg$logFC >= 1]  = "up"
deg$sig[deg$P.Value <= 0.05 & deg$logFC <= -1] = "down"

write.csv(deg, file = "BMDM_rnaseq_group_0.05_1.csv", row.names = TRUE)

  res_df$ID %in% c("mmu04142", "mmu04144", "mmu04145", "mmu04979", "mmu00600", "mmu00531", "mmu00511") ~ "Lysosome / lipid metabolism",
  res_df$ID %in% c("mmu04010", "mmu04068", "mmu04151", "mmu04141")                                      ~ "ROS / stress",
  res_df$ID %in% c("mmu04512", "mmu04510", "mmu05205", "mmu04350")                                      ~ "ECM / CAF activation",

    KEGG_kk,
