sc.QC.Mouse = function(merged_seurat){
  ### 1.1 Number of genes detected per UMI
  merged_seurat$log10GenesPerUMI <- log10(merged_seurat$nFeature_RNA) / log10(merged_seurat$nCount_RNA)
  merged_seurat$feature_ratio    <- merged_seurat$nFeature_RNA / merged_seurat$nCount_RNA
  
  ### 1.2 QC index calculation
  ## 1.2.1 Mitochondrial ratio
  mito_genes = grep("^MT-", rownames(merged_seurat), ignore.case = TRUE, value = TRUE)
  print("mito_genes:")
  print(mito_genes)
  merged_seurat <- PercentageFeatureSet(object = merged_seurat, features = mito_genes, col.name = "percent_mito")
  print(fivenum(merged_seurat@meta.data$percent_mito))
  
  ## 1.2.2 Ribosomal ratio
  ribo_genes = grep("^Rp[sl]", rownames(merged_seurat), ignore.case = TRUE, value = TRUE)
  print("ribo_genes:")
  print(ribo_genes)
  merged_seurat <- PercentageFeatureSet(object = merged_seurat, features = ribo_genes, col.name = "percent_ribo")
  print(fivenum(merged_seurat@meta.data$percent_ribo))
  
  ## 1.2.3 Erythrocyte ratio
  hb_genes <- grep("^Hb[^(p)]", rownames(merged_seurat), ignore.case = TRUE, value = TRUE)
  print("hb_genes:")
  print(hb_genes)
  merged_seurat <- PercentageFeatureSet(object = merged_seurat, features = hb_genes, col.name = "percent_hb")
  print(fivenum(merged_seurat@meta.data$percent_hb))
  
  return(merged_seurat)
}

## 6.2 样本水平细胞比例统计检验
celltype_sample_prop = umap_df %>%
  count(orig.ident, Group, celltype, name = "n_cells") %>%
  complete(nesting(orig.ident, Group), celltype, fill = list(n_cells = 0)) %>%
  group_by(orig.ident) %>%
  mutate(proportion = n_cells / sum(n_cells) * 100) %>%
  ungroup()

celltype_group_prop = celltype_sample_prop %>%
  group_by(Group, celltype) %>%
  summarise(mean_prop = mean(proportion), sd_prop = sd(proportion), n_sample = n(), .groups = "drop")

stat_results_pos = celltype_sample_prop %>%
  group_by(celltype) %>%
  summarise(p.value = t.test(proportion ~ Group)$p.value, .groups = "drop") %>%
  mutate(
    p.adjust     = p.adjust(p.value, method = "BH"),
    significance = case_when(
      p.adjust < 0.001 ~ "***",
      p.adjust < 0.01  ~ "**",
      p.adjust < 0.05  ~ "*",
      TRUE             ~ "ns"
    )
  ) %>%
  left_join(
    celltype_group_prop %>%
      group_by(celltype) %>%
      summarise(y.position = max(mean_prop + sd_prop, na.rm = TRUE) * 1.15, .groups = "drop"),
    by = "celltype"
  )

p.GroupCompare = ggplot(celltype_group_prop, aes(celltype, mean_prop, fill = Group)) +
  geom_col(position = position_dodge(0.8), width = 0.7) +
  geom_errorbar(aes(ymin = pmax(mean_prop - sd_prop, 0), ymax = mean_prop + sd_prop),
                position = position_dodge(0.8), width = 0.25) +
  geom_text(data = stat_results_pos, aes(celltype, y.position, label = significance),
            inherit.aes = FALSE, size = 5, fontface = "bold") +
  scale_fill_manual(values = c(KO = "#E41A1C", WT = "#377EB8")) +
  coord_cartesian(clip = "off") +
  labs(x = NULL, y = "Cell proportion (%)", title = "Cell type composition comparison") +
  theme_classic(base_size = 13) +

    axis.text.x     = element_text(angle = 45, hjust = 1, size = 10),

    legend.position = "top",

  )
